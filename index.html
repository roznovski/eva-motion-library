/* Trace-path для завантаження: точки контуру серця, перевпорядковані
   так, щоб малювання йшло з нижньої вершини за годинниковою стрілкою */
const total = outline.getTotalLength();
const pts = [];
for (let i = 0; i < 160; i++) pts.push(outline.getPointAtLength(total * i / 160));

let tipIndex = 0;
for (let i = 1; i < pts.length; i++) if (pts[i].y > pts[tipIndex].y) tipIndex = i;

const fromTip = pts.slice(tipIndex).concat(pts.slice(0, tipIndex));
const clockwise = [fromTip[0]].concat(fromTip.slice(1).reverse());

const trace = document.createElementNS('http://www.w3.org/2000/svg', 'path');
trace.setAttribute('class', 'trace-path');
trace.setAttribute('d', 'M ' + clockwise.map(p => p.x + ' ' + p.y).join(' L ') + ' Z');
outline.parentNode.insertBefore(trace, outline);

const len = trace.getTotalLength();
trace.style.setProperty('--len', len);
trace.style.setProperty('--neg-len', -len);  // без calc() у кейфреймі — деякі браузери
                                              // не інтерполюють calc(var()) плавно
trace.style.strokeDasharray = len + ' ' + len;

struct WishlistHeart: View {
    @State private var isLiked = false
    @State private var isLoading = false
    @State private var scale: CGFloat = 1

    let pressHaptic = UIImpactFeedbackGenerator(style: .medium)
    let successHaptic = UINotificationFeedbackGenerator()

    var body: some View {
        ZStack {
            if isLoading {
                // кільце розміром під іконку замінює трейс контуру
                Circle()
                    .trim(from: 0, to: 0.25)
                    .stroke(Color(hex: "BFBFBF"), lineWidth: 1.5)
                    .frame(width: 24, height: 24)
                    .rotationEffect(.degrees(isLoading ? 360 : 0))
                    .animation(
                        .linear(duration: 0.8).repeatForever(autoreverses: false),
                        value: isLoading
                    )
            }
            Image(systemName: isLiked ? "heart.fill" : "heart")
                .font(.system(size: 24))
                .foregroundStyle(isLiked ? Color(hex: "212121") : Color(hex: "BFBFBF"))
                .scaleEffect(scale)
        }
        .frame(width: 56, height: 56)
        .onTapGesture { handleTap() }
    }

    func handleTap() {
        if isLiked {
            isLiked = false
            withAnimation(.easeIn(duration: 0.15)) { scale = 0.85 }
            withAnimation(.easeOut(duration: 0.15).delay(0.15)) { scale = 1 }
            return
        }

        pressHaptic.impactOccurred()
        withAnimation(.easeOut(duration: 0.1)) { scale = 0.9 }
        isLoading = true

        Task {
            do {
                try await api.addToWishlist(productId)
                await MainActor.run {
                    isLoading = false
                    isLiked = true
                    successHaptic.notificationOccurred(.success)
                    withAnimation(.spring(response: 0.3, dampingFraction: 0.55)) {
                        scale = 1.25
                    }
                    withAnimation(.spring(response: 0.3, dampingFraction: 0.55).delay(0.22)) {
                        scale = 1
                    }
                }
            } catch {
                await MainActor.run {
                    isLoading = false
                    // shake / toast — сценарій "Обробка помилок"
                }
            }
        }
    }
}
@Composable
fun WishlistHeart(productId: String) {
    var isLiked by remember { mutableStateOf(false) }
    var isLoading by remember { mutableStateOf(false) }
    val scale = remember { Animatable(1f) }
    val haptic = LocalHapticFeedback.current
    val scope = rememberCoroutineScope()

    Box(contentAlignment = Alignment.Center, modifier = Modifier.size(56.dp)) {
        if (isLoading) {
            // кільце розміром під іконку замінює трейс контуру
            CircularProgressIndicator(
                color = Color(0xFFBFBFBF),
                strokeWidth = 1.5.dp,
                modifier = Modifier.size(24.dp)
            )
        }
        Icon(
            imageVector = if (isLiked) Icons.Filled.Favorite else Icons.Outlined.FavoriteBorder,
            contentDescription = "Додати в обране",
            tint = if (isLiked) Color(0xFF212121) else Color(0xFFBFBFBF),
            modifier = Modifier
                .size(24.dp)
                .scale(scale.value)
                .clickable(
                    interactionSource = remember { MutableInteractionSource() },
                    indication = null
                ) {
                    scope.launch {
                        if (isLiked) {
                            isLiked = false
                            scale.animateTo(0.85f, tween(150))
                            scale.animateTo(1f, tween(150))
                            return@launch
                        }

                        haptic.performHapticFeedback(HapticFeedbackType.LongPress)
                        scale.animateTo(0.9f, tween(100))
                        isLoading = true

                        try {
                            api.addToWishlist(productId)
                            isLoading = false
                            isLiked = true
                            haptic.performHapticFeedback(HapticFeedbackType.LongPress)
                            scale.animateTo(
                                1.25f,
                                spring(Spring.DampingRatioMediumBouncy, Spring.StiffnessMedium)
                            )
                            scale.animateTo(1f, tween(220))
                        } catch (e: Exception) {
                            isLoading = false
                            // shake / snackbar — сценарій "Обробка помилок"
                        }
                    }
                }
        )
    }
}

function wireRadioRow(row, onSelect) {
  const wrap = row.querySelector('.checkbox-wrap');
  const btn = row.querySelector('.checkbox-btn');
  const svg = row.querySelector('svg');
  const outline = row.querySelector('.outline');
  const clipCircle = row.querySelector('.clip-circle');
  let active = false;
  let pending = false;

  btn.addEventListener('mouseenter', () => {
    if (active || pending || !supportsHover) return;
    outline.setAttribute('stroke', '#212121');
    svg.style.transform = 'scale(1.06)';
  });
  btn.addEventListener('mouseleave', () => {
    if (active || pending || !supportsHover) return;
    outline.setAttribute('stroke', '#BFBFBF');
    svg.style.transform = 'scale(1)';
  });

  btn.addEventListener('click', () => {
    if (pending || active) return;      // radio: повторний клік по вже обраному нічого не робить

    pending = true;
    svg.style.transform = 'scale(0.95)';
    wrap.classList.add('loading');

    api.selectOption(itemId).then(() => {
      pending = false;
      active = true;
      wrap.classList.remove('loading');
      outline.setAttribute('stroke', '#212121');
      clipCircle.setAttribute('r', '10');
      svg.style.transform = 'scale(1.06)';
      setTimeout(() => (svg.style.transform = 'scale(1)'), 150);
      onSelect();                        // групова ексклюзивність — знімає інші
    });
  });

  return {
    deselect() {                         // викликає група, коли обрали інший варіант
      if (!active) return;
      active = false;
      outline.setAttribute('stroke', '#BFBFBF');
      clipCircle.style.transition = 'r .15s ease-in';
      clipCircle.setAttribute('r', '0');
    }
  };
}

struct RadioRow: View {
    let isSelected: Bool
    let onSelect: () -> Void
    @State private var isLoading = false
    @State private var scale: CGFloat = 1

    let pressHaptic = UIImpactFeedbackGenerator(style: .light)

    var body: some View {
        ZStack {
            if isLoading {
                Circle()
                    .trim(from: 0, to: 0.25)
                    .stroke(Color(hex: "BFBFBF"), lineWidth: 1.5)
                    .frame(width: 20, height: 20)
                    .rotationEffect(.degrees(isLoading ? 360 : 0))
                    .animation(.linear(duration: 0.6).repeatForever(autoreverses: false), value: isLoading)
            } else {
                Image(systemName: isSelected ? "largecircle.fill.circle" : "circle")
                    .font(.system(size: 20))
                    .foregroundStyle(isSelected ? Color(hex: "212121") : Color(hex: "BFBFBF"))
                    .scaleEffect(scale)
            }
        }
        .onTapGesture {
            guard !isSelected else { return }   // radio: не реагує на вже обраний
            pressHaptic.impactOccurred()
            withAnimation(.easeOut(duration: 0.1)) { scale = 0.95 }
            isLoading = true

            Task {
                try? await api.selectOption(itemId)     // реальний виклик
                await MainActor.run {
                    isLoading = false
                    onSelect()                            // батьківська View знімає інші
                    withAnimation(.spring(response: 0.2, dampingFraction: 0.6)) { scale = 1.06 }
                    withAnimation(.spring(response: 0.2, dampingFraction: 0.6).delay(0.15)) { scale = 1 }
                }
            }
        }
    }
}

    setTimeout(() => dot.remove(), 480);

function makeDraggable(thumb, getMin, getMax, setValue, isMinThumb) {
  thumb.addEventListener('pointerdown', (e) => {
    thumb.setPointerCapture(e.pointerId);
    thumb.classList.add('grabbing');
    slider.classList.add('dragging');            // вимикає transition на час drag

    function move(ev) {
      const rect = track.getBoundingClientRect();
      let px = ev.clientX - rect.left;
      px = Math.max(12, Math.min(rect.width - 12, px));
      if (isMinThumb) px = Math.min(px, getMax() - 16);
      else px = Math.max(px, getMin() + 16);
      setValue(px);                               // 1:1 з курсором, без анімації
    }
    function up() {
      thumb.classList.remove('grabbing');
      slider.classList.remove('dragging');        // повертає spring-transition для наступних стрибків
      thumb.removeEventListener('pointermove', move);
      thumb.removeEventListener('pointerup', up);
    }
    thumb.addEventListener('pointermove', move);
    thumb.addEventListener('pointerup', up);
  });
}

struct RangeSlider: View {
    @State private var minValue: CGFloat = 0.0
    @State private var maxValue: CGFloat = 0.5
    @State private var draggingMin = false
    @State private var draggingMax = false

    let trackWidth: CGFloat = 288

    var body: some View {
        ZStack(alignment: .leading) {
            Capsule().fill(Color(hex: "EBEBEB")).frame(height: 8)
            Capsule().fill(Color(hex: "76BF44"))
                .frame(width: (maxValue - minValue) * trackWidth, height: 8)
                .offset(x: minValue * trackWidth)

            thumb(value: $minValue, isDragging: $draggingMin, otherValue: maxValue, isMin: true)
            thumb(value: $maxValue, isDragging: $draggingMax, otherValue: minValue, isMin: false)
        }
        .frame(width: trackWidth, height: 24)
    }

    func thumb(value: Binding<CGFloat>, isDragging: Binding<Bool>, otherValue: CGFloat, isMin: Bool) -> some View {
        Circle()
            .fill(.white)
            .frame(width: 22, height: 22)
            .overlay(Circle().stroke(isDragging.wrappedValue ? Color(hex: "76BF44") : Color(hex: "D8D8D8"), lineWidth: 2))
            .scaleEffect(isDragging.wrappedValue ? 1.2 : 1)
            .shadow(color: .black.opacity(isDragging.wrappedValue ? 0.22 : 0), radius: 4, y: 2)
            .offset(x: value.wrappedValue * trackWidth - trackWidth / 2)
            // Під час активного drag — без withAnimation, рух 1:1 з пальцем.
            // Пружний withAnimation лишається тільки для програмних стрибків.
            .animation(isDragging.wrappedValue ? nil : .spring(response: 0.25, dampingFraction: 0.7), value: value.wrappedValue)
            .gesture(
                DragGesture(minimumDistance: 0)
                    .onChanged { g in
                        isDragging.wrappedValue = true
                        let raw = (g.location.x + trackWidth / 2) / trackWidth
                        value.wrappedValue = isMin
                            ? min(max(0, raw), otherValue - 0.06)
                            : max(min(1, raw), otherValue + 0.06)
                    }
                    .onEnded { _ in
                        isDragging.wrappedValue = false
                        withAnimation(.spring(response: 0.25, dampingFraction: 0.7)) {}
                    }
            )
    }
}

fun RangeSlider() {

                .height(8.dp)
                .clip(RoundedCornerShape(4.dp))
                .background(Color(0xFF76BF44))
        )

        listOf(true, false).forEach { isMin ->
            val value = if (isMin) minValue else maxValue
            // Під час drag — offset рахується напряму з пальця (1:1, без анімації).
            // animateFloatAsState залишається тільки для стрибків по кліку на трек.

                            if (isMin) minValue = raw.coerceAtMost(maxValue - 0.06f)
                            else maxValue = raw.coerceAtLeast(minValue + 0.06f)
                        }
                    }
            )
        }
    }

function burst(wrap) {
  const n = 8, dist = 22;
  for (let i = 0; i < n; i++) {
    const angle = (i / n) * Math.PI * 2;
    const dot = document.createElement('span');
    dot.style.cssText = 'position:absolute;left:50%;top:50%;width:4px;height:4px;border-radius:50%;background:#76BF44;pointer-events:none;transform:translate(-50%,-50%);opacity:1;transition:transform .5s cubic-bezier(.2,.8,.3,1), opacity .5s ease-out;';
    wrap.appendChild(dot);
    void dot.offsetWidth;         // forced reflow — без нього transition не спрацьовує з першого кліку
    dot.style.transform = 'translate(calc(-50% + ' + Math.cos(angle) * dist + 'px), calc(-50% + ' + Math.sin(angle) * dist + 'px))';
    dot.style.opacity = '0';
    setTimeout(() => dot.remove(), 550);
  }
}

btn.addEventListener('click', () => {
  if (pending || inCart) return;

  pending = true;
  btn.style.transform = 'scale(0.97)';     // press-фідбек, <100ms
  setTimeout(() => (btn.style.transform = 'scale(1)'), 100);

  if (!slowToggle.checked) {
    succeed(); return;                       // оптимістичний UI — без стану завантаження
  }

  wrap.classList.add('loading');
  status.textContent = 'Додаємо до кошика…';
  api.addToCart(productId).then(() => {      // реальний виклик
    wrap.classList.remove('loading');
    succeed();
  });
});

function succeed() {
  pending = false; inCart = true;
  btn.classList.add('in-cart');             // запускає CSS-transition bg + color
  label.textContent = 'У кошику';
  btn.style.transform = 'scale(1.04)';     // spring-bounce при успіху
  setTimeout(() => (btn.style.transform = 'scale(1)'), 150);
  burst(wrap);                             // 8 зелених частинок
}
struct AddToCartButton: View {
    let productId: String
    @State private var inCart = false
    @State private var isLoading = false
    @State private var scale: CGFloat = 1

    let pressHaptic = UIImpactFeedbackGenerator(style: .medium)
    let successHaptic = UINotificationFeedbackGenerator()

    var body: some View {
        Button { handleTap() } label: {
            HStack(spacing: 8) {
                if isLoading {
                    // Дуга, що обертається — замість іконки під час очікування
                    Circle()
                        .trim(from: 0, to: 0.25)
                        .stroke(Color.white, lineWidth: 1.5)
                        .frame(width: 20, height: 20)
                        .rotationEffect(.degrees(isLoading ? 360 : 0))
                        .animation(
                            .linear(duration: 0.8).repeatForever(autoreverses: false),
                            value: isLoading
                        )
                } else {
                    Image(systemName: inCart ? "cart.badge.checkmark" : "cart")
                        .font(.system(size: 16))
                }
                Text(inCart ? "У кошику" : "До кошика")
                    .font(.system(size: 14, weight: .semibold))
            }
            .foregroundColor(inCart ? Color(hex: "212121") : .white)
            .frame(height: 40)
            .padding(.horizontal, 16)
            // Фон і обведення переходять через withAnimation нижче
            .background(inCart ? Color.white : Color(hex: "212121"))
            .overlay(
                RoundedRectangle(cornerRadius: 8)
                    .stroke(Color(hex: "212121"), lineWidth: 1.5)
            )
            .cornerRadius(8)
            .scaleEffect(scale)
        }
        .buttonStyle(.plain)
        .disabled(inCart)
    }

    func handleTap() {
        pressHaptic.impactOccurred()
        withAnimation(.easeOut(duration: 0.1)) { scale = 0.97 }
        withAnimation(.easeOut(duration: 0.1).delay(0.1)) { scale = 1 }
        isLoading = true

        Task {
            try? await api.addToCart(productId)     // реальний виклик
            await MainActor.run {
                isLoading = false
                successHaptic.notificationOccurred(.success)
                // Анімує зміну фону й тексту разом зі spring-bounce
                withAnimation(.easeInOut(duration: 0.2)) { inCart = true }
                withAnimation(.spring(response: 0.25, dampingFraction: 0.65)) {
                    scale = 1.04
                }
                withAnimation(.spring(response: 0.25, dampingFraction: 0.65).delay(0.15)) {
                    scale = 1
                }
            }
        }
    }
}
@Composable
fun AddToCartButton(productId: String) {
    var inCart by remember { mutableStateOf(false) }
    var isLoading by remember { mutableStateOf(false) }
    val scale = remember { Animatable(1f) }
    val haptic = LocalHapticFeedback.current
    val scope = rememberCoroutineScope()

    // Плавна зміна кольорів при переході в стан "У кошику"
    val bgColor by animateColorAsState(
        targetValue = if (inCart) Color.White else Color(0xFF212121),
        animationSpec = tween(durationMillis = 200),
        label = "cartBg"
    )
    val contentColor by animateColorAsState(
        targetValue = if (inCart) Color(0xFF212121) else Color.White,
        animationSpec = tween(durationMillis = 200),
        label = "cartContent"
    )

    Row(
        horizontalArrangement = Arrangement.spacedBy(8.dp),
        verticalAlignment = Alignment.CenterVertically,
        modifier = Modifier
            .height(40.dp)
            .clip(RoundedCornerShape(8.dp))
            .background(bgColor)
            .border(1.5.dp, Color(0xFF212121), RoundedCornerShape(8.dp))
            .scale(scale.value)
            .clickable(
                enabled = !inCart,
                interactionSource = remember { MutableInteractionSource() },
                indication = null
            ) {
                scope.launch {
                    haptic.performHapticFeedback(HapticFeedbackType.LongPress)
                    scale.animateTo(0.97f, tween(100))       // press-фідбек

                    isLoading = true
                    api.addToCart(productId)                  // реальний виклик
                    isLoading = false
                    inCart = true                             // запускає animateColorAsState

                    haptic.performHapticFeedback(HapticFeedbackType.LongPress)
                    scale.animateTo(                          // spring-bounce при успіху
                        1.04f,
                        spring(Spring.DampingRatioMediumBouncy, Spring.StiffnessMedium)
                    )
                    scale.animateTo(1f, tween(150))
                }
            }
            .padding(horizontal = 16.dp)
    ) {
        if (isLoading) {
            CircularProgressIndicator(
                color = Color.White,
                strokeWidth = 1.5.dp,
                modifier = Modifier.size(20.dp)
            )
        } else {
            Icon(
                imageVector = if (inCart)
                    Icons.Outlined.ShoppingCartCheckout
                else
                    Icons.Outlined.ShoppingCart,
                contentDescription = null,
                tint = contentColor,
                modifier = Modifier.size(20.dp)
            )
        }
        Text(
            text = if (inCart) "У кошику" else "До кошика",
            color = contentColor,
            fontSize = 14.sp,
            fontWeight = FontWeight.SemiBold
        )
    }
}
